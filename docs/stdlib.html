re #

python
import re

m = re.match(r"\d+", "123abc")
if m:
    print(m.group(0))     # "123"

parts = re.split(r",\s*", "a, b, c")
out   = re.sub(r"\d", "#", "a1b2")  # "a#b#"
all_n = re.findall(r"\d+", "1 22 333")

pat = re.compile(r"(\w+)\s+(\w+)")
m2  = pat.search("hello world")
print(m2.group(1), m2.group(2))

Supports: ., *, +, ?, |, ^, $, [...], \d\w\s and their uppercase complements, {n,m}, (?:...) non-capturing groups, capturing groups (...). Flags: re.IGNORECASE, re.MULTILINE, re.DOTALL.

csv

python
import csv

lines: list[str] = ["name,age", "alice,30", "bob,25"]
rows: list[list[str]] = csv.reader(lines)
for row in rows:
    print(row[0], row[1])

line: str = csv.writer_row(["alice", "30"])

dr = csv.DictReader(lines)
data_row: list[str] = dr.rows[0]
print(dr.get(data_row, "name"))

csv.reader(lines) returns list[list[str]] — each row is a plain list of fields. csv.writer_row(fields) formats one row. csv.writer_rows(rows) formats a list[list[str]]. DictReader parses headers.

collections #

python
from collections import deque, Counter, defaultdict, OrderedDict

dq = deque([1, 2, 3])
dq.appendleft(0)
dq.popleft()

cnt = Counter(["a", "b", "a", "c", "a"])
print(cnt["a"])      # 3
top = cnt.most_common(2)

dd = defaultdict(int)
dd["k"] += 1

od = OrderedDict()
od["a"] = 1
od.move_to_end("a")

Also available: namedtuple, ChainMap, UserDict, UserList, UserString.

asmlib.hardware #

python
from asmlib import hardware

hardware.console.clear()
hardware.console.set_cursor(x=0, y=0)
hardware.console.set_color(fg=0x0A, bg=0x00)
hardware.console.print_at(5, 5, "Hello!")

info = hardware.cpu.info()
print(info.vendor, info.cores)
tsc = hardware.cpu.rdtsc()

hardware.memory.write8(0xB8000, 0x41)
val = hardware.memory.read8(0xB8000)

hardware.port.out8(0x3F8, 0x41)
byte = hardware.port.in8(0x3F8)

Low-level hardware access for freestanding targets. Wraps direct NASM instructions for console I/O, CPU feature detection, memory-mapped I/O, and port-mapped I/O.

asmlib.gui #

python
from asmlib import gui

win = gui.Window("My App", 800, 600)
win.set_icon("icon.ico")

while win.is_open():
    ev = win.poll_event()
    win.clear(0x1E1E2E)
    win.draw_rect(10, 10, 100, 50, 0xFF4444)
    win.draw_text("Hello GUI", 20, 20, 0xFFFFFF)
    win.present()

Win32 native window with software renderer (no SDL/Qt). Draw primitives: draw_rect, draw_circle, draw_line, draw_text, draw_image. Events: keyboard, mouse, resize, close.

Surfaces & window icons

python
import gui

# Load a BMP surface (returns an opaque handle)
surf = gui.load_bmp("icon.bmp")

# Set the window's taskbar / title-bar icon from a surface
gui.set_window_icon(win, surf)

# Free the surface when no longer needed
gui.free_surface(surf)
FunctionDescription
gui.load_bmp(path)Load a BMP file from disk; returns a surface handle (int)
gui.set_window_icon(win, surf)Set the Win32 window icon from a surface handle
gui.free_surface(surf)Release the surface memory
The high-level Window.set_icon(path) method accepts an .ico path directly and handles load + set internally.
